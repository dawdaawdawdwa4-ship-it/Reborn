local woodToolPreference = "Auto best axe"

local function savePreferences()
    if type(writefile) ~= "function" then return false end
    local data = {
        HarvestCrop = CONFIG.HarvestCrop, PlantType = CONFIG.PlantType,
        AutoReplant = CONFIG.AutoReplant, MineRange = CONFIG.MineRange,
        WoodRange = CONFIG.WoodRange, MineDelay = CONFIG.MineDelay,
        MineScanInterval = CONFIG.MineScanInterval, PlantDelay = CONFIG.PlantDelay,
        WoodToolPreference = woodToolPreference, MineToolPreference = mineToolPreference,
    }
    local ok = pcall(function() writefile(SETTINGS_FILE, HttpService:JSONEncode(data)) end)
    return ok
end
local function loadPreferences()
    if type(readfile) ~= "function" or type(isfile) ~= "function" then return false end
    local ok, data = pcall(function()
        if not isfile(SETTINGS_FILE) then return nil end
        return HttpService:JSONDecode(readfile(SETTINGS_FILE))
    end)
    if not ok or type(data) ~= "table" then return false end
    if type(data.HarvestCrop) == "string" then CONFIG.HarvestCrop = data.HarvestCrop end
    if type(data.PlantType) == "string" then CONFIG.PlantType = data.PlantType end
    if type(data.AutoReplant) == "boolean" then CONFIG.AutoReplant = data.AutoReplant end
    if type(data.MineRange) == "number" then CONFIG.MineRange = math.clamp(data.MineRange, 15, 150) end
    if type(data.WoodRange) == "number" then CONFIG.WoodRange = math.clamp(data.WoodRange, 15, 150) end
    if type(data.MineDelay) == "number" then CONFIG.MineDelay = math.clamp(data.MineDelay, 0.08, 1) end
    if type(data.MineScanInterval) == "number" then CONFIG.MineScanInterval = math.clamp(data.MineScanInterval, 0.15, 2) end
    if type(data.PlantDelay) == "number" then CONFIG.PlantDelay = math.clamp(data.PlantDelay, 0.08, 1) end
    if type(data.WoodToolPreference) == "string" then woodToolPreference = data.WoodToolPreference end
    if type(data.MineToolPreference) == "string" then mineToolPreference = data.MineToolPreference end
    return true
end

local function applyFishingMode(modeName)
    local profile = FISHING_MODES[modeName]
    if not profile then return end
    activeFishingMode = modeName
    CONFIG.FishingPollInterval = profile.poll
    CONFIG.CastTimeout = profile.castTimeout
    CONFIG.ReelTimeout = profile.reelTimeout
    CONFIG.ReelSettleDelay = profile.settle
    CONFIG.ActionCooldown = profile.cooldown
    for name, modeButton in pairs(modeButtons) do
        modeButton.BackgroundColor3 = name == modeName and COLORS.accent or COLORS.panel
    end
    modeHeading.Text = "FISHING MODE  ·  " .. string.upper(modeName)
    if modeName == "Safe" then
        modeInfo.Text = "Safe: slower polling and longer waits; best for high latency or checking reliability."
    elseif modeName == "Balanced" then
        modeInfo.Text = "Balanced: responsive timing with extra catch-confirmation time for everyday use."
    elseif modeName == "Fast" then
        modeInfo.Text = "Fast: shortest waits and quickest retry; may count a slow server response as a miss."
    else
        modeInfo.Text = "Test: conservative confirmation timing for comparing catch rate against Fast mode. Reset Stats between runs."
    end
end

local function applyBait()
    local character = player.Character
    local baitTool = character and character:FindFirstChild("bait")
    if not baitTool then
        local backpack = player:FindFirstChildOfClass("Backpack")
        baitTool = backpack and backpack:FindFirstChild("bait")
    end
    if not baitTool then
        setStatus("Auto Bait: bait tool not found")
        return false
    end
    local ok, result = pcall(function()
        local net = game:GetService("ReplicatedStorage"):FindFirstChild("rbxts_include")
        local nodeModules = net and net:FindFirstChild("node_modules")
        local rbxts = nodeModules and nodeModules:FindFirstChild("@rbxts")
        local netFolder = rbxts and rbxts:FindFirstChild("net")
        local outFolder = netFolder and netFolder:FindFirstChild("out")
        local remoteGroup = outFolder and outFolder:FindFirstChild("ULrMeokpcNr7uD")
        local remote = remoteGroup and remoteGroup:FindFirstChild("raqRasAwcny1zh")
        assert(remote and remote:IsA("RemoteFunction"), "bait remote unavailable")
        return remote:InvokeServer({
            TrNhTTgExIb1Bg = "06105b98-6d2f-40fb-9979-fbd1b925fe6daa81356f-b501-468d-80f0-5871006ce00a",
            TlSR8Mzd9FLwaZ = { tool = baitTool },
        })
    end)
    if not ok then setStatus("Auto Bait failed: " .. tostring(result)) end
    return ok
end

local function setBaitEnabled(state)
    baitEnabled = state
    baitCycleId += 1
    local myCycle = baitCycleId
    baitToggle.Text = baitEnabled and "AUTO BAIT ON  [120s]" or "AUTO BAIT OFF  [120s]"
    baitToggle.BackgroundColor3 = baitEnabled and COLORS.on or COLORS.off
    if not baitEnabled then
        setStatus("Auto Bait stopped")
        return
    end
    task.spawn(function()
        while baitEnabled and baitCycleId == myCycle and gui.Parent do
            local applied = applyBait()
            if applied then setStatus("Auto Bait applied; refresh in 120s", COLORS.good) end
            for remaining = 120, 1, -1 do
                if not baitEnabled or baitCycleId ~= myCycle or not gui.Parent then return end
                baitToggle.Text = string.format("AUTO BAIT ON  [%ds]", remaining)
                task.wait(1)
            end
        end
    end)
end

local stats

local function newStats()
	return {
		casts = 0,
		bites = 0,
		catches = 0,
		missed = 0,
		streak = 0,
		bestStreak = 0,
		harvests = 0,
		lastMineName = "–",
		activeTime = 0,
		onSince = nil,
		perFish = {},   -- [name] = count
		recent = {},    -- { {name=, at=}, ... } newest first
	}
end

stats = newStats()

local function getActiveTime()
	local t = stats.activeTime
	if stats.onSince then
		t += os.clock() - stats.onSince
	end
	return t
end

local function formatTime(seconds)
	seconds = math.floor(seconds)
	return string.format("%02d:%02d:%02d",
		math.floor(seconds / 3600),
		math.floor((seconds % 3600) / 60),
		seconds % 60)
end

local function formatAgo(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s ago"
	elseif seconds < 3600 then
		return math.floor(seconds / 60) .. "m ago"
	end
	return math.floor(seconds / 3600) .. "h ago"
end

local function getTopFish()
	local bestName, bestCount = nil, 0
	for name, count in pairs(stats.perFish) do
		if count > bestCount then
			bestName, bestCount = name, count
		end
	end
	if bestName then
		return bestName .. " ×" .. bestCount
	end
	return "–"
end

local function updateBoard()
	local t = getActiveTime()

	rows.harvests.Text = tostring(stats.harvests)
	rows.mineRate.Text = t >= 5
		and string.format("%.0f", stats.harvests / (t / 60))
		or "–"
	rows.lastMine.Text = stats.lastMineName or "–"
	rows.catches.Text = tostring(stats.catches)
	rows.missed.Text = tostring(stats.missed)
	rows.casts.Text = tostring(stats.casts)
	rows.bites.Text = tostring(stats.bites)
	rows.rate.Text = stats.bites > 0
		and string.format("%d%%", math.floor(stats.catches / stats.bites * 100 + 0.5))
		or "–"
    local rateText = stats.bites > 0 and string.format("%d%%", math.floor(stats.catches / stats.bites * 100 + 0.5)) or "–"
    statsOverlayText.Text = string.format("Catches %d  |  Misses %d\nCasts %d  |  Bites %d\nRate %s  |  Mode %s", stats.catches, stats.missed, stats.casts, stats.bites, rateText, string.upper(activeFishingMode))
	rows.streak.Text = stats.streak .. " / " .. stats.bestStreak
	rows.perHour.Text = t >= 30
		and string.format("%.0f", stats.catches / (t / 3600))
		or "–"
	rows.time.Text = formatTime(t)

	local now = os.clock()
	for i, slot in ipairs(recentLabels) do
		local entry = stats.recent[i]
		if entry then
			slot.name.Text = "🐟 " .. entry.name
			slot.name.TextColor3 = COLORS.text
			slot.time.Text = formatAgo(now - entry.at)
		else
			slot.name.Text = i == 1 and "Nothing yet" or ""
			slot.name.TextColor3 = COLORS.muted
			slot.time.Text = ""
		end
	end
end

	lbl.TextColor3 = Color3.new(1, 1, 1)
	lbl.TextSize = 15
	TweenService:Create(lbl, TweenInfo.new(0.4, Enum.EasingStyle.Quad), {
		TextColor3 = color,

	}):Play()
end

local function registerCatch(fishName)
	fishName = fishName or "Fish"

	stats.catches += 1
	stats.streak += 1
	stats.perFish[fishName] = (stats.perFish[fishName] or 0) + 1

	table.insert(stats.recent, 1, { name = fishName, at = os.clock() })
	if #stats.recent > CONFIG.RecentCount then
		table.remove(stats.recent)
	end

	if stats.streak > stats.bestStreak then
		stats.bestStreak = stats.streak
		if stats.bestStreak > 1 then
			flash(rows.streak, COLORS.gold)
		end
	end

	flash(rows.catches, COLORS.good)
	updateBoard()
end

local function registerMiss()
	stats.missed += 1
	stats.streak = 0
	flash(rows.missed, COLORS.bad)
	updateBoard()
end

local function registerHarvest(name)
	stats.harvests += 1
	if name then
		stats.lastMineName = name
	end
	if stats.harvests % 10 == 0 then
		flash(rows.harvests, COLORS.harvest)
	end
	updateBoard()
end

--------------------------------------------------------------------
-- New-item detection (used to name the fish you caught)
-- Rods moving between Backpack and Character are the same instance,
-- so only genuinely new instances count.
--------------------------------------------------------------------
local knownItems = setmetatable({}, { __mode = "k" })
local lastNewItem = nil -- { name = string, at = number }

local function markKnown(container)
	for _, child in ipairs(container:GetChildren()) do
		knownItems[child] = true
	end
end

local function onItemAdded(child)
	if not knownItems[child] then
		knownItems[child] = true
		lastNewItem = { name = child.Name, at = os.clock() }
	end
end

local function hookBackpack(backpack)
	markKnown(backpack)
	track(backpack.ChildAdded:Connect(onItemAdded))
end

local currentBackpack = player:FindFirstChildOfClass("Backpack")
if currentBackpack then
	hookBackpack(currentBackpack)
end

track(player.ChildAdded:Connect(function(child)
	if child:IsA("Backpack") then
		hookBackpack(child)
	end
end))

if player.Character then
	markKnown(player.Character)
end
track(player.CharacterAdded:Connect(markKnown))

-- Window dragging (single handler; mouse + touch; keep the title reachable)
--------------------------------------------------------------------
local dragging = false
local dragStart, startPosition
local dragInputType

local function beginDrag(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = frame.Position
        dragInputType = input.UserInputType
    end
end

track(title.InputBegan:Connect(beginDrag))
if titleBar and titleBar ~= title then
    track(titleBar.InputBegan:Connect(beginDrag))
end

track(UserInputService.InputChanged:Connect(function(input)
    if not dragging or not dragStart or not startPosition then return end
    local isMouseMove = dragInputType == Enum.UserInputType.MouseButton1
        and input.UserInputType == Enum.UserInputType.MouseMovement
    local isTouchMove = dragInputType == Enum.UserInputType.Touch
        and input.UserInputType == Enum.UserInputType.Touch
    if not isMouseMove and not isTouchMove then return end

    local delta = input.Position - dragStart
    local camera = workspace.CurrentCamera
    local viewport = camera and camera.ViewportSize or Vector2.new(1280, 720)
    local scale = (uiScale and uiScale.Scale) or 1
    local visibleW = WIDTH * scale
    local visibleH = (minimized and MINI_HEIGHT or FULL_HEIGHT) * scale
    local x = startPosition.X.Offset + delta.X
    local y = startPosition.Y.Offset + delta.Y
    x = math.clamp(x, 8 - visibleW, math.max(8, viewport.X - 40))
    y = math.clamp(y, 0, math.max(0, viewport.Y - math.min(40, visibleH)))
    frame.Position = UDim2.new(startPosition.X.Scale, x, startPosition.Y.Scale, y)
end))

track(UserInputService.InputEnded:Connect(function(input)
    if (input.UserInputType == Enum.UserInputType.MouseButton1
        and dragInputType == Enum.UserInputType.MouseButton1)
        or (input.UserInputType == Enum.UserInputType.Touch
        and dragInputType == Enum.UserInputType.Touch) then
        dragging = false
        dragStart = nil
        startPosition = nil
        dragInputType = nil
    end
end))

-- Note: VirtualInputManager is normally blocked in a standard LocalScript.
local function clickOnce()
	local camera = workspace.CurrentCamera
	if not camera then
		return false
	end

	local x = camera.ViewportSize.X / 2
	local y = camera.ViewportSize.Y / 2

	local pressed = pcall(function()
		VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, nil, 0)
	end)
	if not pressed then
		-- Standard LocalScripts commonly cannot use VirtualInputManager.
		return false
	end
	task.wait(0.01)
	-- Always attempt release independently so an input error cannot leave the
	-- simulated mouse button held down.
	local released = pcall(function()
		VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, nil, 0)
	end)
	return released
end

local function waitForTug(bobber, myCycle)
	local tugged = false

	local function checkTug()
		local value = bobber:GetAttribute("Tug")
		if value and value ~= false then
			tugged = true
		end
	end

	checkTug()
	local conn = bobber:GetAttributeChangedSignal("Tug"):Connect(checkTug)

	while running(myCycle) and bobber.Parent and not tugged do
		task.wait(CONFIG.FishingPollInterval)
	end

	conn:Disconnect()
	return tugged and running(myCycle)
end

-- Polls until predicate() is true, the cycle stops, or timeout. Returns predicate result.
local function waitUntil(myCycle, timeout, predicate)
	local deadline = os.clock() + timeout
	while running(myCycle) and os.clock() < deadline do
		if predicate() then
			return true
		end
		task.wait(CONFIG.FishingPollInterval)
	end
	-- A stopped/restarted cycle must not report a stale success after cancellation.
	return running(myCycle) and predicate()
end

		plantRemote:InvokeServer({
			TrNhTTgExIb1Bg = HARVEST_UUID,

-- Plant a grid of crops around the player
local function plantAroundPlayer(blockType)
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then
		return 0
	end
	blockType = blockType or CONFIG.PlantType
	local origin = hrp.Position
	local planted = 0
	local r = CONFIG.PlantRadius
	local step = CONFIG.PlantSpacing

	for x = -r, r, step do
		for z = -r, r, step do
			local pos = Vector3.new(origin.X + x, origin.Y, origin.Z + z)
			-- Keep upright rotation like the Cobalt sample
			local cf = CFrame.new(pos) * CFrame.Angles(0, 0, 0)
			if firePlant(cf, blockType) then
				planted += 1
			end
			task.wait(CONFIG.PlantDelay)
		end
	end
	return planted
end

		"woodAxe,,,,,,,",
		"stoneAxe,,,,,,,",
		"ironAxe,,,,,,,",
		"ironPickaxe,,,,,,,",
		"stonePickaxe,,,,,,,",
		"woodPickaxe,,,,,,,",
	}
	for _, layout in ipairs(layouts) do
		if setHotbarLayout(layout) then
			break
		end
	end
end

local function equipBestResourceTool(category)
    local character = player.Character
    local backpack = player:FindFirstChildOfClass("Backpack")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end

    local preference = category == "wood" and woodToolPreference or mineToolPreference
    local wantAxe = category == "wood"
    local best, bestScore = nil, -math.huge
    local containers = {character, backpack}
    for _, container in ipairs(containers) do
        if container then
            for _, tool in ipairs(container:GetChildren()) do
                if tool:IsA("Tool") then
                    local name = string.lower(tool.Name)
                    local isPick = string.find(name, "pickaxe", 1, true) ~= nil
                    local isAxe = string.find(name, "axe", 1, true) ~= nil and not isPick
                    if (wantAxe and isAxe) or (not wantAxe and isPick) then
                        local score = 1
                        if string.find(name, "mythic", 1, true) then score = 100
                        elseif string.find(name, "diamond", 1, true) then score = 90
                        elseif string.find(name, "gold", 1, true) then score = 80
                        elseif string.find(name, "steel", 1, true) then score = 70
                        elseif string.find(name, "iron", 1, true) then score = 60
                        elseif string.find(name, "stone", 1, true) then score = 40
                        elseif string.find(name, "wood", 1, true) then score = 20 end
                        if not string.find(string.lower(preference), "auto best", 1, true) then
                            local wanted = string.lower(preference):gsub(" axe", ""):gsub(" pickaxe", "")
                            if not string.find(name, wanted, 1, true) then
                                score = -math.huge -- manual preference means exact material only
                            else
                                score = score + 1000
                            end
                        end
                        if score > bestScore then best, bestScore = tool, score end
                    end
                end
            end
        end
    end
    if best and bestScore > -math.huge then
        local ok, err = pcall(function() humanoid:EquipTool(best) end)
        if not ok then
            if setStatus then setStatus("Tool equip failed: " .. tostring(err), COLORS.bad) end
            return false
        end
        return true
    end
    if setStatus then
        setStatus((category == "wood" and "No matching axe: " or "No matching pickaxe: ") .. tostring(preference), COLORS.warn)
    end
    return false
end

local function equipPickaxe()
    return equipBestResourceTool(woodEnabled and "wood" or "mine")
end

-- Mining / resource name patterns (stone, iron, coal, ores, rocks, trees...)
-- This replaces the separate Auto Wood Farmer target scanner/loop: tree targets
-- are included here and use the same GUI, F7 toggle, stats, and cleanup.
-- All of these use the BLOCK remote (OEzzgCCBzocxJw)
local MINE_PATTERNS = {
	-- ores & rocks (hub + island)
	"rock", "Rock", "stone", "Stone", "ore", "Ore",
	"iron", "Iron", "coal", "Coal", "gold", "Gold",
	"copper", "Copper", "tin", "Tin", "silver", "Silver",
	"crystal", "Crystal", "gem", "Gem", "mineral", "Mineral",
	"rockStone", "rockIron", "rockCoal", "rockGold",
	-- trees / wood
	"tree", "Tree", "wood", "Wood", "log", "Log",
	"oak", "Oak", "pine", "Pine", "birch", "Birch",
	"treeOrange", "treeApple", "treePalm",
}

-- Crop / bush models use the MODEL remote (hFuV0Z1w601mXP)
local MODEL_PATTERNS = {
	"berry", "Berry", "berryBush", "bush", "Bush",
	"wheat", "Wheat",
	"crop", "Crop",
	"flower", "Flower",
	"carrot", "potato", "corn", "tomato",
	"pumpkin", "melon", "cabbage", "lettuce",
	"berryHarvested",
}

-- Never harvest/mine these
local SKIP_PATTERNS = {
	"sapling", "Sapling", "seedling", "Seedling",
}

local function matchesAny(name, patterns)
	local lower = string.lower(name)
	for _, p in ipairs(patterns) do
		if string.find(lower, string.lower(p), 1, true) then
			return true
		end
	end
	return false
end

local function getPlayerIsland()
	local islands = workspace:FindFirstChild("Islands")
	if not islands then
		return nil
	end
	-- Prefer the island closest to the player
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if hrp then
		local best, bestDist = nil, math.huge
		for _, island in ipairs(islands:GetChildren()) do
			if island:IsA("Model") or island:IsA("Folder") then
				local blocks = island:FindFirstChild("Blocks")
				if blocks then
					for _, child in ipairs(blocks:GetChildren()) do
						local pos
						if child:IsA("BasePart") then
							pos = child.Position
						elseif child:IsA("Model") then
							pos = child:GetPivot().Position
						end
						if pos then
							local d = (pos - hrp.Position).Magnitude
							if d < bestDist and d < 250 then
								bestDist = d
								best = island
							end
						end
					end
				end
			end
		end
		if best then
			return best
		end
	end
	-- fallback
	for _, island in ipairs(islands:GetChildren()) do
		if string.find(island.Name, tostring(player.UserId), 1, true)
			or string.find(string.lower(island.Name), "island", 1, true) then
			return island
		end
	end
	return islands:GetChildren()[1]
end

local function getPartPosition(obj)
	if obj:IsA("BasePart") then
		return obj.Position
	elseif obj:IsA("Model") then
		local ok, pivot = pcall(function()
			return obj:GetPivot().Position
		end)
		if ok and pivot then
			return pivot
		end
	end
	local part = obj:FindFirstChildWhichIsA("BasePart", true)
	if part then
		return part.Position
	end
	return nil
end

local function targetCategory(name)
    if matchesAny(name, MODEL_PATTERNS) then return "crop" end
    local lower = string.lower(name)
    if string.find(lower, "tree", 1, true) or string.find(lower, "wood", 1, true)
        or string.find(lower, "log", 1, true) or string.find(lower, "oak", 1, true)
        or string.find(lower, "pine", 1, true) or string.find(lower, "birch", 1, true)
        or string.find(lower, "palm", 1, true) then return "wood" end
    return "mine"
end

local function modeAllows(name)
    local category = targetCategory(name)
    if category == "crop" then
        local n, wanted = string.lower(name), string.lower(CONFIG.HarvestCrop)
        return harvestEnabled and (wanted == "berrybush" and string.find(n, "berry", 1, true) ~= nil or string.find(n, wanted, 1, true) ~= nil)
    elseif category == "wood" then
        return woodEnabled
    end
    return mineEnabled
end

local function getMineTargets()
	local targets = {}
	local seenTargets = {}
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then
		return targets
	end
	local myPos = hrp.Position
	local range = CONFIG.MineRange

	-- 1) PRIMARY: exact path from Cobalt
	--    workspace.spawnPrefabs.regenBlocks.hubRocks.Blocks.rockStone
	local function scanHubBlocks(blocksFolder)
		if not blocksFolder then
			return
		end
		for _, obj in ipairs(blocksFolder:GetChildren()) do
			-- Mine EVERY child in hub rocks (rockStone, rockIron, rockCoal, etc.)
			local pos = getPartPosition(obj)
			if pos and (pos - myPos).Magnitude <= range and not seenTargets[obj] and modeAllows(obj.Name) then
				seenTargets[obj] = true
				table.insert(targets, { kind = "block", instance = obj, name = obj.Name })
			end
		end
	end

	local ok, hubBlocks = pcall(function()
		return workspace.spawnPrefabs.regenBlocks.hubRocks.Blocks
	end)
	if ok and hubBlocks then
		scanHubBlocks(hubBlocks)
	end

	-- Also scan any other areas under regenBlocks
	local ok2, regen = pcall(function()
		return workspace.spawnPrefabs.regenBlocks
	end)
	if ok2 and regen then
		for _, area in ipairs(regen:GetChildren()) do
			if area.Name ~= "hubRocks" then
				local blocks = area:FindFirstChild("Blocks")
				if blocks then
					scanHubBlocks(blocks)
				end
			end
		end
	end

	-- 2) ALL island Blocks + Prefabs (stone, trees, berryBush, wheat, crops)
	local islands = workspace:FindFirstChild("Islands")
	if islands then
		for _, island in ipairs(islands:GetChildren()) do
			local foldersToScan = {}
			local blocksFolder = island:FindFirstChild("Blocks")
			if blocksFolder then
				table.insert(foldersToScan, blocksFolder)
			end
			local prefabs = island:FindFirstChild("Prefabs")
			if prefabs then
				table.insert(foldersToScan, prefabs)
			end

			for _, folder in ipairs(foldersToScan) do
				for _, obj in ipairs(folder:GetChildren()) do
					local name = obj.Name
					-- Never touch saplings / seedlings
					if matchesAny(name, SKIP_PATTERNS) then
						continue
					end
					local pos = getPartPosition(obj)
					-- crops get a slightly larger range
					local maxDist = matchesAny(name, MODEL_PATTERNS) and (range + 40) or range
					if pos and (pos - myPos).Magnitude <= maxDist and modeAllows(name) then
						if matchesAny(name, MODEL_PATTERNS) then
							-- berryBush, wheat, crops → MODEL remote
							table.insert(targets, { kind = "model", instance = obj, name = name })
						elseif matchesAny(name, MINE_PATTERNS) then
							-- stone, trees, ores → BLOCK remote
							table.insert(targets, { kind = "block", instance = obj, name = name })
						end
					end
				end
			end
		end
	end

	-- Process the nearest resource/tree first (from Auto Wood Farmer).
	local hrpNow = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
	if hrpNow then
		table.sort(targets, function(a, b)
			local pa = getPartPosition(a.instance)
			local pb = getPartPosition(b.instance)
			if not pa then return false end
			if not pb then return true end
			return (pa - hrpNow.Position).Magnitude < (pb - hrpNow.Position).Magnitude
		end)
	end
	-- Nearby-first ordering improves throughput without sending parallel requests.
	table.sort(targets, function(a, b)
		local pa = getPartPosition(a.instance)
		local pb = getPartPosition(b.instance)
		if not pa then return false end
		if not pb then return true end
		return (pa - myPos).Magnitude < (pb - myPos).Magnitude
	end)
	return targets
end

local function harvestLoop(myCycle)
	if not resolveRemotes() then
		setStatus("Mine remotes not found – re-spy with Cobalt", COLORS.bad)
		return
	end

	while harvestCycleId == myCycle and (harvestEnabled or woodEnabled or mineEnabled) and gui.Parent ~= nil do
		local targets = getMineTargets()
		if #targets == 0 then
			-- Debug: check if path exists
			local pathOk = pcall(function()
				return workspace.spawnPrefabs.regenBlocks.hubRocks.Blocks
			end)
			if pathOk then
				setStatus("Mine: hub found, no rocks in range", COLORS.warn)
			else
				setStatus("Mine: hub path missing – stand near rocks", COLORS.warn)
			end
			task.wait(CONFIG.MineScanInterval)
		else
			-- Sequential mining – one rock at a time so the server grants items
			for _, t in ipairs(targets) do
				if harvestCycleId ~= myCycle or not (harvestEnabled or woodEnabled or mineEnabled) then
					break
				end

				-- Skip if already destroyed
				if not t.instance or not t.instance.Parent then
					continue
				end

				-- Equip the matching tool for this target before interacting.
                local category = targetCategory(t.name)
                if category == "wood" then
                    equipBestResourceTool("wood")
                elseif category == "mine" then
                    equipBestResourceTool("mine")
                end

				-- Remember position before harvest (for replant)
				local harvestPos = getPartPosition(t.instance)

				local ok
				if t.kind == "block" then
					ok = fireBlockMine(t.instance)
				else
					ok = fireModelHarvest(t.instance)
				end

				if ok then
					-- Wait briefly and check if the target was removed
					task.wait(0.08)
					local destroyed = not t.instance.Parent
					registerHarvest(t.name)
					if t.kind == "model" then
						setStatus("🌾 " .. t.name .. "  ×" .. stats.harvests, COLORS.good)
						-- Replant at the same spot after harvesting crops
						if CONFIG.PlantEnabled and CONFIG.AutoReplant and harvestPos and targetCategory(t.name) == "crop" then
							local plantType = CONFIG.PlantType
							local cf = CFrame.new(harvestPos)
							if firePlant(cf, plantType) then
								setStatus("🌱 planted " .. plantType, COLORS.good)
							end
						end
					elseif destroyed then
						setStatus("⛏ " .. t.name .. "  ×" .. stats.harvests, COLORS.harvest)
					else
						setStatus("⛏ hitting " .. t.name, COLORS.muted)
					end
				end

				task.wait(CONFIG.MineDelay)
			end
		end
	end

	if harvestCycleId == myCycle and not (harvestEnabled or woodEnabled or mineEnabled) then
		if not enabled then
			setStatus("Idle")
		end
	end
end

-- Main fishing loop
--------------------------------------------------------------------
local setEnabled -- forward declaration
local setHarvestEnabled

local function run(myCycle)
	local failedCasts = 0

	while running(myCycle) do
		local bobber = getBobber()

		if bobber then
			failedCasts = 0
			setStatus("Waiting for fish...")
			local tugged = waitForTug(bobber, myCycle)
			if not running(myCycle) then break end

			if not tugged then
				setStatus("Bobber disappeared")
				task.wait(0.1)
			elseif os.clock() - lastAction >= CONFIG.ActionCooldown then
				lastAction = os.clock()
				local reelTime = os.clock()
				stats.bites += 1
				updateBoard()
				setStatus("Tug! Reeling in...", COLORS.gold)
				clickOnce()
				task.wait(CONFIG.ReelSettleDelay)
				setStatus("Waiting for catch...")
				local gone = waitUntil(myCycle, CONFIG.ReelTimeout, function()
					return getBobber() == nil
				end)
				if not running(myCycle) then break end
				if gone then
					task.wait(0.05)
					local fishName
					if lastNewItem and lastNewItem.at >= reelTime then fishName = lastNewItem.name end
					registerCatch(fishName)
					setStatus("Caught " .. (fishName or "a fish") .. "! 🐟", COLORS.good)
				else
					registerMiss()
					setStatus("Reel not confirmed - retrying", COLORS.warn)
					task.wait(0.05)
				end
			end
		elseif not hasRodEquipped() then
			setStatus("Equip your fishing rod", COLORS.warn)
			task.wait(0.5)
		else
			if os.clock() - lastAction >= CONFIG.ActionCooldown then
				lastAction = os.clock()
				setStatus("Casting...")
				if clickOnce() then
					stats.casts += 1
					updateBoard()
					-- Only wait for a bobber after an actual cast input; this avoids
					-- burning the full timeout on frames where the cooldown is active.
					local found = waitUntil(myCycle, CONFIG.CastTimeout, function()
						return getBobber() ~= nil
					end)
					if not running(myCycle) then break end
					if not found then
						failedCasts += 1
						if failedCasts >= CONFIG.MaxFailedCasts then
							setEnabled(false)
							setStatus("Stopped: no bobber after " .. failedCasts .. " casts", COLORS.bad)
							return
						end
						setStatus(string.format("No bobber (%d/%d) - retrying", failedCasts, CONFIG.MaxFailedCasts), COLORS.warn)
						 task.wait(0.03)
					end
				else
					setStatus("Input blocked - run in a supported client", COLORS.bad)
					task.wait(0.25)
				end
			else
				task.wait(0.02)
			end
		end
	end

	if cycleId == myCycle and not enabled then
		if not (harvestEnabled or woodEnabled or mineEnabled) then
			setStatus("Idle")
		end
	end
end

setEnabled = function(state)
	if state == enabled then
		return
	end

	enabled = state
	cycleId += 1

	if enabled then
		if not stats.onSince then
			stats.onSince = os.clock()
		end
		toggle.Text = "FISH ON  [" .. CONFIG.ToggleKey.Name .. "]"
		toggle.BackgroundColor3 = COLORS.on
		setStatus("Fishing started...")
		task.spawn(run, cycleId)
	else
		toggle.Text = "FISH OFF  [" .. CONFIG.ToggleKey.Name .. "]"
		toggle.BackgroundColor3 = COLORS.off
		if not (harvestEnabled or woodEnabled or mineEnabled) and stats.onSince then
			stats.activeTime += os.clock() - stats.onSince
			stats.onSince = nil
		end
		if not (harvestEnabled or woodEnabled or mineEnabled) then
			setStatus("Idle")
		end
	end

	updateBoard()
end

local function refreshFarmButtons()
    harvestToggle.Text = harvestEnabled and "CROP FARM ON" or "CROP FARM OFF"
    harvestToggle.BackgroundColor3 = harvestEnabled and COLORS.on or COLORS.off
    woodToggle.Text = woodEnabled and "WOOD FARM ON" or "WOOD FARM OFF"
    woodToggle.BackgroundColor3 = woodEnabled and COLORS.on or COLORS.off
    mineToggle.Text = mineEnabled and "MINING ON" or "MINING OFF"
    mineToggle.BackgroundColor3 = mineEnabled and COLORS.on or COLORS.off
end

local function setFarmMode(mode, state)
    -- Resolve the requested mode explicitly. The old chained `and/or`
    -- expression could read another mode's state when this mode was off.
    local current
    if mode == "crop" then
        current = harvestEnabled
    elseif mode == "wood" then
        current = woodEnabled
    elseif mode == "mine" then
        current = mineEnabled
    else
        return
    end
    if current == state then return end
    if mode == "crop" then harvestEnabled = state
    elseif mode == "wood" then woodEnabled = state
    else mineEnabled = state end
    harvestCycleId += 1
    if harvestEnabled or woodEnabled or mineEnabled then
        if not stats.onSince then stats.onSince = os.clock() end
        ensurePickaxeLayout()
        equipPickaxe()
        setStatus(mode == "crop" and "Crop farming started" or mode == "wood" and "Wood farming started" or "Mining started", COLORS.good)
        task.spawn(harvestLoop, harvestCycleId)
    else
        if not enabled and stats.onSince then
            stats.activeTime += os.clock() - stats.onSince
            stats.onSince = nil
        end
        if not enabled then setStatus("Idle") end
    end
    refreshFarmButtons()
    refreshModeStatus()
    updateBoard()
    savePreferences()
end

-- Real expanding dropdowns matching the crop selector's visual style.
local function makeToolDropdown(page, anchor, options, kind, yPosition, onChoose)
    local menu = Instance.new("Frame")
    menu.Name = kind .. "Dropdown"
    menu.Size = UDim2.new(1, -12, 0, #options * 27)
    menu.Position = UDim2.fromOffset(6, yPosition)
    menu.BackgroundColor3 = COLORS.panel
    menu.BorderSizePixel = 0
    menu.Visible = false
    menu.ClipsDescendants = true
    menu.ZIndex = 25
    menu.Parent = page
    corner(menu, 6)
    local list = Instance.new("UIListLayout")
    list.Padding = UDim.new(0, 1)
    list.SortOrder = Enum.SortOrder.LayoutOrder
    list.Parent = menu
    for i, optionName in ipairs(options) do
        local opt = button({Parent = menu, Text = optionName, TextSize = 11,
            Size = UDim2.new(1, -4, 0, 26), Position = UDim2.new(), Bg = COLORS.panel})
        opt.LayoutOrder = i
        opt.ZIndex = 26
        track(opt.Activated:Connect(function()
            onChoose(i)
            anchor.Text = kind .. ": " .. optionName .. "  ▼"
            menu.Visible = false
        end))
    end
    track(anchor.Activated:Connect(function()
        menu.Visible = not menu.Visible
    end))
end

    mineRangeBtn.Text = "Mining range: " .. CONFIG.MineRange .. " studs  (+ / −)"
    savePreferences()
end))

track(delayButton.Activated:Connect(function()
    CONFIG.MineDelay = CONFIG.MineDelay >= 0.40 and 0.08 or math.min(0.40, CONFIG.MineDelay + 0.04)
    delayButton.Text = "Mine delay: " .. string.format("%.2fs", CONFIG.MineDelay) .. "  (click to adjust)"
    savePreferences()
end))
track(scanButton.Activated:Connect(function()
    CONFIG.MineScanInterval = CONFIG.MineScanInterval >= 1.0 and 0.15 or math.min(1.0, CONFIG.MineScanInterval + 0.15)
    scanButton.Text = "Scan interval: " .. string.format("%.2fs", CONFIG.MineScanInterval) .. "  (click to adjust)"
    savePreferences()
end))
track(saveButton.Activated:Connect(function()
    local saved = savePreferences()
    saveButton.Text = saved and "Preferences saved ✓" or "Save unavailable in this environment"
    saveButton.BackgroundColor3 = saved and COLORS.on or COLORS.warn
end))

track(UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then
		return
	end
	if input.KeyCode == CONFIG.ToggleKey then
		setEnabled(not enabled)
	elseif input.KeyCode == CONFIG.HarvestKey then
		setHarvestEnabled(not harvestEnabled)
	end
end))

local minimized = false
track(minimize.Activated:Connect(function()
	minimized = not minimized
	minimize.Text = minimized and "+" or "–"
	TweenService:Create(frame, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
		Size = UDim2.fromOffset(WIDTH, minimized and MINI_HEIGHT or FULL_HEIGHT),
	}):Play()
end))

track(reset.Activated:Connect(function()
	local wasOn = stats.onSince ~= nil
	stats = newStats()
	if wasOn then
		stats.onSince = os.clock()
	end
	updateBoard()
end))

track(layoutBtn.Activated:Connect(function()
	-- Full layout from your Cobalt capture
	local ok = setHotbarLayout(
		"woodPickaxe,,sapling,hoe,berryHarvested,ironOre,saplingOrange,yellowFlowerSeeds"
	)
	if ok then
		equipPickaxe()
		setStatus("Hotbar layout applied", COLORS.good)
	else
		setStatus("Layout remote failed – re-spy", COLORS.bad)
	end
end))

track(plantBtn.Activated:Connect(function()
	setStatus("🌱 Planting " .. CONFIG.PlantType .. "...", COLORS.good)
	task.spawn(function()
		local n = plantAroundPlayer(CONFIG.PlantType)
		setStatus("🌱 Planted " .. n .. " × " .. CONFIG.PlantType, COLORS.good)
	end)
end))

--------------------------------------------------------------------
-- Live refresh + cleanup
--------------------------------------------------------------------
updateBoard()

task.spawn(function()
	while gui.Parent do
		updateBoard()
		task.wait(1)
	end
end)

gui.Destroying:Connect(function()
    baitEnabled = false
    baitCycleId += 1
	enabled = false
	harvestEnabled = false
	cycleId += 1
	harvestCycleId += 1
	for _, conn in ipairs(connections) do
		conn:Disconnect()
	end
	table.clear(connections)
end)
