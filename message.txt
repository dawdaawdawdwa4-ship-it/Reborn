    function catalog.FindPath(path, expectedClass)
        local ok, found = pcall(function()
            local parts = string.split(path, ".")
            local node = game:FindFirstChild(parts[1])
            for i = 2, #parts do
                node = node and node:FindFirstChild(parts[i])
                if not node then return nil end
            end
            if expectedClass and node and not node:IsA(expectedClass) then return nil end
            return node
        end)
        return ok and found or nil
    end
    function catalog.FindNamed(name, expectedClass)
        local base = "ReplicatedStorage.rbxts_include.node_modules.@rbxts.net.out"
        local exact = catalog.FindPath(base .. "._NetManaged." .. name, expectedClass)
        if exact then return exact end
        local out = catalog.FindPath(base)
        if not out then return nil end
        for _, child in ipairs(out:GetDescendants()) do
            if child.Name == name and child:IsA(expectedClass) then return child end
        end
        return nil
    end

local function serialize(value, seen, budget, depth)
    budget = budget or { count = 0, complete = true }
    seen = seen or {}
    depth = depth or 0
    budget.count += 1
    if depth > 20 or budget.count > 500 then
        budget.complete = false
        return "nil --[[capture limit]]", budget.complete
    end
    local kind = typeof(value)
    if kind == "nil" or kind == "boolean" then return tostring(value), budget.complete end
    if kind == "number" then
        if value ~= value then return "(0/0)", budget.complete end
        if value == math.huge then return "math.huge", budget.complete end
        if value == -math.huge then return "-math.huge", budget.complete end
        return string.format("%.17g", value), budget.complete
    end
    if kind == "string" then
        if #value > 4096 then budget.complete = false end
        return string.format("%q", string.sub(value, 1, 4096)), budget.complete
    end
    if kind == "Instance" then
        local path, complete = instancePath(value)
        if not complete then budget.complete = false end
        return path, budget.complete
    end
    if kind == "Vector3" then
        return string.format("Vector3.new(%.17g, %.17g, %.17g)", value.X, value.Y, value.Z), budget.complete
    end
    if kind == "Vector2" then
        return string.format("Vector2.new(%.17g, %.17g)", value.X, value.Y), budget.complete
    end
    if kind == "CFrame" then
        local components = { value:GetComponents() }
        for i, component in ipairs(components) do components[i] = string.format("%.17g", component) end
        return "CFrame.new(" .. table.concat(components, ", ") .. ")", budget.complete
    end
    if kind == "Color3" then
        return string.format("Color3.new(%.17g, %.17g, %.17g)", value.R, value.G, value.B), budget.complete
    end
    if kind == "EnumItem" then return tostring(value), budget.complete end
    if kind == "table" then
        if seen[value] then budget.complete = false; return "nil --[[cyclic table]]", false end
        seen[value] = true
        local parts = {}
        for key, item in next, value do
            if budget.count > 500 then budget.complete = false; break end
            local encodedKey = serialize(key, seen, budget, depth + 1)
            local encodedValue = serialize(item, seen, budget, depth + 1)
            table.insert(parts, "[" .. encodedKey .. "] = " .. encodedValue)
        end
        seen[value] = nil
        return "{" .. table.concat(parts, ", ") .. "}", budget.complete
    end
    budget.complete = false
    return "nil --[[unsupported: " .. kind .. "]]", false
end

                and type(args[1].TrNhTTgExIb1Bg) == "string" then
                capture.RequestIds[remote.Name] = args[1].TrNhTTgExIb1Bg
            end

            if not capture.Active then return end
            local budget = { count = 0, complete = true }
            local path, validPath = instancePath(remote)
            if not validPath then budget.complete = false end
            local encoded = {}
            for i = 1, args.n do encoded[i] = serialize(args[i], {}, budget) end
            local record = string.format("-- Call %d | %s | t=%.3f | %s\n%s:%s(%s)",
                #capture.Records + 1, remote.Name, os.clock(), budget.complete and "COMPLETE" or "INCOMPLETE",

                warn("[Reborn capture] 100-call limit reached. Press F9 to export.")
            end
        end)
    end

local mineNames = {"rock", "stone", "ore", "iron", "coal", "gold", "copper", "tin", "silver", "crystal", "gem", "mineral"}
local readiness = {"Mature", "IsMature", "FullyGrown", "Harvestable", "ReadyToHarvest"}
local function contains(name, patterns)
    for _, pattern in ipairs(patterns) do
        if string.find(name, pattern, 1, true) then return true end
    end
    return false
end
function FarmLogic.Category(name)
    name = string.lower(name)
    if contains(name, {"sapling", "seedling", "seed", "harvested"}) then return nil end
    if contains(name, cropNames) then return "crop" end
    if contains(name, woodNames) then return "wood" end
    if contains(name, mineNames) then return "mine" end
    return nil
end
function FarmLogic.Fruits(model)
    local fruits={}
    if FarmLogic.Category(model.Name)~="wood" then return fruits end
    for _,stage in ipairs(model:GetChildren()) do
        if string.match(stage.Name,"^stage%-%d+$") then
            local fruit=stage:FindFirstChild("Fruit")
            if fruit and FarmLogic.IsReady(fruit) then table.insert(fruits,fruit) end
        end
    end
    return fruits
end
function FarmLogic.CropMatches(name, selected)
    local wanted = string.lower(selected)
    if wanted == "berrybush" then wanted = "berry" end
    return FarmLogic.Category(name) == "crop" and string.find(string.lower(name), wanted, 1, true) ~= nil
end
function FarmLogic.Range(category, config)
    if category == "wood" then return config.WoodRange end
    if category == "crop" then return config.CropRange end
    return config.MineRange
end
function FarmLogic.Snapshot(target)
    local values = {}
    for _, name in ipairs(readiness) do values[name] = target:GetAttribute(name) end
    return values
end
function FarmLogic.IsReady(target)
    for _, name in ipairs(readiness) do
        if target:GetAttribute(name) == false then return false end
    end
    return true -- Unknown growth schema: do not invent a numeric maturity threshold.
end
function FarmLogic.Confirmed(target, before, prompt)
    if not target.Parent then return true end
    for _, name in ipairs(readiness) do
        if before[name] == true and target:GetAttribute(name) == false then return true end
    end
    return prompt ~= nil and (not prompt.Parent or not prompt.Enabled)
end
function FarmLogic.HarvestPrompt(target)
    for _, child in ipairs(target:GetDescendants()) do
        if child:IsA("ProximityPrompt") and child.Enabled then
            local text = string.lower(child.Name .. " " .. child.ActionText)
            if contains(text, {"harvest", "høst", "collect", "pick"}) then return child end
        end
    end
    return nil
end
function FarmLogic.HoldPrompt(prompt, active, waitFor, now)
    local began = pcall(function() prompt:InputHoldBegin() end)
    if not began then pcall(function() prompt:InputHoldEnd() end); return false end
    local held = pcall(function()
        local deadline = now() + prompt.HoldDuration
        while active() and now() < deadline do waitFor(0.05) end
    end)
    local released = pcall(function() prompt:InputHoldEnd() end)
    return held and released and active()
end
return FarmLogic
end)()

local AutoBuild = {}
function AutoBuild.Plan(preset, width, length, height)
    width, length, height = math.floor(width), math.floor(length), math.floor(height)
    assert(width >= 1 and width <= 12 and length >= 1 and length <= 12 and height >= 1 and height <= 8, "Use sizes 1–12 and height 1–8")
    local cells = {}
    local function add(x,y,z,role)
        table.insert(cells, {x=x-math.floor(width/2),y=y,z=z+1,role=role or "main"})
        assert(#cells <= 300, "Plan exceeds 300 blocks; reduce its dimensions")
    end
    if preset == "Platform" then
        for z=0,length-1 do for x=0,width-1 do add(x,0,z) end end
    elseif preset == "Farm plots" then
        for z=0,length-1 do
            for x=0,width-1 do add(x,1,z,x%3==2 and "path" or "main") end
        end
    elseif preset == "Wall" then
        for y=1,height do for x=0,width-1 do add(x,y,0) end end
    elseif preset == "Stairs" then
        for z=0,length-1 do
            for y=1,z+1 do for x=0,width-1 do add(x,y,z) end end
        end
    else error("Unknown blueprint") end
    table.sort(cells,function(a,b)
        if a.y ~= b.y then return a.y < b.y end
        if a.z ~= b.z then return a.z < b.z end
        return a.x < b.x
    end)
    return cells
end
function AutoBuild.new(ctx)

    local function characterRoot()

    local function blockSize()

    end
    local function occupied(cf,size)
        local util=ctx.GameModule("ReplicatedStorage.TS.util.block-utils","BlockUtils")
        if util and type(util.getBlocksFromLocation)=="function" then
            local ok,blocks=pcall(function() return util.getBlocksFromLocation(cf.Position,{},1,true) end)
            if ok and type(blocks)=="table" then return #blocks>0 end
        end
        local params=OverlapParams.new()
        params.FilterType=Enum.RaycastFilterType.Exclude
        local char=ctx.Player.Character
        local ignored={}
        if char then table.insert(ignored,char) end
        if self.PreviewFolder then table.insert(ignored,self.PreviewFolder) end
        params.FilterDescendantsInstances=ignored
        for _,part in ipairs(space:GetPartBoundsInBox(cf,Vector3.new(size-.2,size-.2,size-.2),params)) do
            if part.CanCollide then return true end
        end
        return false
    end
    function self:Preview()
        if self.Running then status("Stop the build before changing its plan.");return false end
        self:ClearPreview()
        local char,root=characterRoot()
        if not root then status("Wait for your character to spawn.");return false end
        local ok,plan=pcall(AutoBuild.Plan,self.Preset,self.Width,self.Length,self.Height)
        if not ok then status(plan);return false end
        local material=inputs.Material.Text:match("^%s*(.-)%s*$")
        local pathMaterial=inputs.Path.Text:match("^%s*(.-)%s*$")
        if material=="" or self.Preset=="Farm plots" and pathMaterial=="" then status("Choose bed/build and path materials.");return false end
        local params=RaycastParams.new()
        params.FilterType=Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances={char}
        local ground=space:Raycast(root.Position,Vector3.new(0,-40,0),params)
        if not ground then status("Stand on solid ground before previewing.");return false end
        local size=blockSize()
        local look=root.CFrame.LookVector
        local yaw=math.round(math.atan2(-look.X,-look.Z)/(math.pi/2))*(math.pi/2)+math.rad(self.Rotation)
        local anchor=CFrame.new(math.round(root.Position.X/size)*size,math.round((ground.Position.Y-size/2)/size)*size,math.round(root.Position.Z/size)*size)*CFrame.Angles(0,yaw,0)
        anchor=anchor*CFrame.new(self.OffsetX*size,self.OffsetY*size,-self.OffsetZ*size)
        self.Anchor=anchor
        local definitions=game:GetService("ReplicatedStorage"):FindFirstChild("Blocks")
        if definitions then
            if not definitions:FindFirstChild(material) then status("Unknown build block: "..material);return false end
            if self.Preset=="Farm plots" and not definitions:FindFirstChild(pathMaterial) then status("Unknown path block: "..pathMaterial);return false end
        end
        local folder=Instance.new("Folder");folder.Name="RebornBuildPreview";folder.Parent=space
        self.PreviewFolder=folder
        self.Cells={}
        for _,cell in ipairs(plan) do
            local cf=anchor*CFrame.new(cell.x*size,cell.y*size,-cell.z*size)
            local part=Instance.new("Part")
            part.Name="PlannedBlock";part.Anchored=true;part.CanCollide=false;part.CanTouch=false;part.CanQuery=false
            part.Size=Vector3.new(size-.15,size-.15,size-.15);part.CFrame=cf
            part.Transparency=.65;part.Material=Enum.Material.ForceField
            part.Color=cell.role=="path" and Color3.fromRGB(210,180,100) or Color3.fromRGB(100,160,255)
            part.Parent=folder
            table.insert(self.Cells,{CFrame=cf,Material=cell.role=="path" and pathMaterial or material,Preview=part})
        end
        self.BlockSize=size
        status(string.format("Preview: %s, %d blocks. Blue = build/beds; gold = paths.",self.Preset,#self.Cells))
        return true
    end
    function self:Move(dx,dy,dz)
        if self.Running then status("Stop the build before moving its blueprint.");return false end
        if not self.Cells and not self:Preview() then return false end
        local anchor=self.Anchor*CFrame.new(dx*self.BlockSize,dy*self.BlockSize,-dz*self.BlockSize)
        local delta=anchor*self.Anchor:Inverse()
        for _,cell in ipairs(self.Cells) do
            cell.CFrame=delta*cell.CFrame
            if cell.Preview then cell.Preview.CFrame=cell.CFrame end
        end
        self.Anchor=anchor
        self.OffsetX+=dx;self.OffsetY+=dy;self.OffsetZ+=dz
        self.MoveLabel.Text=string.format("Offset: right %d / up %d / forward %d blocks",self.OffsetX,self.OffsetY,self.OffsetZ)
        status("Blueprint moved. Start builds the visible preview positions.")
        return true
    end
    function self:Start()
        if self.Running then return end
        if not self.Cells and not self:Preview() then return end
        self.Token+=1
        local token=self.Token
        self.Running=true;buttons.Start.Text="Building…"
        if self.MoveMenu then self.MoveMenu.Visible=false end
        local function active() return self.Running and self.Token==token and ctx.Alive() end
        scheduler.spawn(function()
            local placed,skipped,failed,streak=0,0,0,0
            local ok,err=pcall(function()
                for index,cell in ipairs(self.Cells) do
                    if not active() then return end
                    local _,root=characterRoot()
                    local reach=ctx.GameModule("ReplicatedStorage.Shared.SharedConfig","CLIENT_REACH")
                    if type(reach)~="number" then reach=20 end
                    while active() and (not root or (root.Position-cell.CFrame.Position).Magnitude>reach) do
                        status(string.format("Move within %d studs of block %d/%d, or press Stop.",reach,index,#self.Cells))
                        scheduler.wait(.2);_,root=characterRoot()
                    end
                    if not active() then return end
                    if occupied(cell.CFrame,self.BlockSize) then
                        skipped+=1
                    else
                        status(string.format("Building %d/%d: %s",index,#self.Cells,cell.Material))
                        local sent,reason=ctx.Place({cframe=cell.CFrame,blockType=cell.Material})
                        if not active() then return end
                        local confirmed=false
                        if sent then
                            local deadline=now()+1
                            repeat
                                if not active() then return end
                                confirmed=occupied(cell.CFrame,self.BlockSize)
                                if confirmed then break end
                                scheduler.wait(.05)
                            until now()>=deadline
                        end
                        if confirmed then placed+=1;streak=0
                        else failed+=1;streak+=1 end
                        if streak>=3 then
                            status("Build paused after 3 failures: "..tostring(reason or "no block appeared")..". Check materials, support and island access.")
                            return
                        end
                    end
                    if cell.Preview then cell.Preview:Destroy();cell.Preview=nil end
                    scheduler.wait(.2)
                end
                if active() then status(string.format("Finished: %d confirmed, %d occupied/skipped, %d failed.",placed,skipped,failed)) end
            end)
            if self.Token==token then
                self.Running=false;buttons.Start.Text="Start build"
                if not ok then status("Build stopped: "..tostring(err)) end
            end
        end)
    end
    function self:Destroy() self:Stop();self:ClearPreview() end
    local function label(text,y,height)

    local function button(name,text,x,y,width,callback)

    label("Build farm beds and paths, platforms, walls or solid stairs.",4,36)

local function loadPreferences()
    if type(readfile) ~= "function" or type(isfile) ~= "function" then return false end
    local ok, data = pcall(function()
        if not isfile(SETTINGS_FILE) then return nil end
        return HttpService:JSONDecode(readfile(SETTINGS_FILE))
    end)
    if not ok or type(data) ~= "table" then return false end
    if type(data.HarvestCrop) == "string" then CONFIG.HarvestCrop = data.HarvestCrop end
    if type(data.PlantType) == "string" then CONFIG.PlantType = data.PlantType end
    if type(data.AutoReplant) == "boolean" then CONFIG.AutoReplant = data.AutoReplant end
    if type(data.FruitEnabled) == "boolean" then CONFIG.FruitEnabled = data.FruitEnabled end
    if type(data.MineRange) == "number" then CONFIG.MineRange = math.clamp(data.MineRange, 15, 150) end
    if type(data.CropRange) == "number" then CONFIG.CropRange = math.clamp(data.CropRange, 10, 120) end
    if type(data.WoodRange) == "number" then CONFIG.WoodRange = math.clamp(data.WoodRange, 15, 150) end
    if type(data.MineDelay) == "number" then CONFIG.MineDelay = math.clamp(data.MineDelay, 0.08, 1) end
    if type(data.MineScanInterval) == "number" then CONFIG.MineScanInterval = math.clamp(data.MineScanInterval, 0.15, 2) end
    if type(data.PlantDelay) == "number" then CONFIG.PlantDelay = math.clamp(data.PlantDelay, 0.08, 1) end
    if type(data.WoodToolPreference) == "string" then woodToolPreference = data.WoodToolPreference end
    if type(data.MineToolPreference) == "string" then mineToolPreference = data.MineToolPreference end
    return true
end

local function createPage(name)
    local tab = hub:Tab({ Name = name })
    tab._label.TextSize = 14
    airflowTabs[name] = tab
    -- Keep the existing farming controls and their callbacks in the Airflow page.
    -- Its stock list/empty state are replaced by our scrollable content.
    tab.List:Destroy()
    for _, child in ipairs(tab._page:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end
    local page = Instance.new("ScrollingFrame")
    page.Name = name .. "Page"
    page.Position = UDim2.fromOffset(18, 58)
    page.Size = UDim2.new(1, -36, 1, -90)
    page.BackgroundTransparency = 1
    page.BorderSizePixel = 0
    page.ScrollBarThickness = 6
    page.VerticalScrollBarInset = Enum.ScrollBarInset.ScrollBar
    page.ElasticBehavior = Enum.ElasticBehavior.WhenScrollable
    page.ScrollBarImageColor3 = COLORS.accent
    page.CanvasSize = UDim2.new()
    page.AutomaticCanvasSize = Enum.AutomaticSize.Y
    page.ScrollingDirection = Enum.ScrollingDirection.Y
    page.Parent = tab._page
    table.insert(hub._controls, page) -- Scrolling/clicking controls must not drag the window.
    -- Airflow's intro reveals its list's children. Use the content as that list.
    tab.List = page
    pages[name] = page
    return page
end

-- BAIT TOOL INPUT BEGIN
RemoteCatalog.ActivateBait = function(tool)
    if baitBusy or fishingInputBusy then return false, "Waiting for current rod input" end
    local character = player.Character
    if not character then return false, "Wait for your character" end
    local previous = character:FindFirstChildOfClass("Tool")
    local token = baitCycleId
    local function active() return baitEnabled and baitCycleId == token and gui.Parent ~= nil and player.Character == character end
    if not active() then return false, "Auto Bait stopped" end
    baitBusy = true
    local activated = false
    local ok, failure = pcall(function()
        if not RemoteCatalog.EquipTool(tool) then error("Could not equip bait") end
        task.wait(0.1)
        if not active() then return end
        if tool.Parent ~= character or character:FindFirstChildOfClass("Tool") ~= tool then error("Bait tool not equipped") end
        tool:Activate()
        activated = true
        task.wait(0.15)
    end)
    -- Restore the rod even after Stop or activation errors; do not overwrite a manual tool change.
    if previous and previous ~= tool and player.Character == character then
        local current = character:FindFirstChildOfClass("Tool")
        if current == tool or current == nil then pcall(RemoteCatalog.EquipTool, previous) end
    end
    baitBusy = false
    if not ok then return false, failure end
    return activated and active(), "tool activation attempted"
end
-- BAIT TOOL INPUT END

local function applyBait()
    local character = player.Character
    local baitTool = character and character:FindFirstChild("bait")
    if not baitTool then
        local backpack = player:FindFirstChildOfClass("Backpack")
        baitTool = backpack and backpack:FindFirstChild("bait")
    end
    if not baitTool then
        baitInfo.Text = "No bait found. Keep bait in your inventory; retrying every 5s."
        baitInfo.TextColor3 = COLORS.warn
        setStatus("Auto Bait: bait tool not found")
        return false, "bait tool not found"
    end
    local ok, result = RemoteCatalog.CallBait(baitTool)
    if not ok then
        baitInfo.Text = "Bait waiting: " .. tostring(result)
        baitInfo.TextColor3 = COLORS.warn
        setStatus("Auto Bait failed: " .. tostring(result))
    else
        baitInfo.Text = result == "tool activation attempted" and "Bait tool activated. Next attempt in 60s; check the game's bait effect."
            or "Bait request accepted. Refreshes every 60 seconds."
        baitInfo.TextColor3 = COLORS.good
    end
    return ok, result
end

local function setBaitEnabled(state)
    if baitEnabled == state then return end
    baitEnabled = state
    baitCycleId += 1
    local myCycle = baitCycleId
    baitToggle.Text = baitEnabled and "AUTO BAIT ON  [60s]" or "AUTO BAIT OFF  [60s]"
    baitToggle.BackgroundColor3 = baitEnabled and COLORS.on or COLORS.off
    if not baitEnabled then
        setStatus("Auto Bait stopped")
        return
    end
    task.spawn(function()
        while baitEnabled and baitCycleId == myCycle and gui.Parent do
            local applied, detail = applyBait()
            if not baitEnabled or baitCycleId ~= myCycle or not gui.Parent then return end
            if applied then setStatus(detail == "tool activation attempted" and "Bait tool activated; next attempt in 60s" or "Auto Bait applied; refresh in 60s", COLORS.good) end
            baitToggle.BackgroundColor3 = applied and COLORS.on or COLORS.off
            local delay = applied and 60 or 5
            for remaining = delay, 1, -1 do
                if not baitEnabled or baitCycleId ~= myCycle or not gui.Parent then return end
                baitToggle.Text = string.format(applied and "AUTO BAIT ON  [%ds]" or "AUTO BAIT ON · RETRY  [%ds]", remaining)
                task.wait(1)
            end
        end
    end)
end

    if not recorder then
        RemoteCatalog.CaptureStatus.Text = RemoteCatalog.CaptureAttempted

            or "Recorder OFF. Farming uses the game API; capture is optional."

        return
    end
    local state = recorder.Stopped and "Stopped" or (recorder.Active and "Recording" or "Paused")

    for _, name in ipairs({"OEzzgCCBzocxJw", "ohy6O10QKzyGZE", "raqRasAwcny1zh", "RZQQNNDRUvfj1N"}) do
        if RemoteCatalog.RequestId(name) then table.insert(learned, RemoteCatalog.VerifiedRequests[name]) end

local stats

local function newStats()
	return {
		casts = 0,
		bites = 0,
		catches = 0,
		missed = 0,
		streak = 0,
		bestStreak = 0,
		harvests = 0,
        resourceCounts = {wood=0,mine=0,crop=0},
        resourceAttempts = {wood=0,mine=0,crop=0},
        resourceLast = {},
		lastMineName = "–",
		activeTime = 0,
		onSince = nil,
		perFish = {},   -- [name] = count
		recent = {},    -- { {name=, at=}, ... } newest first
	}
end

stats = newStats()

local function getActiveTime()
	local t = stats.activeTime
	if stats.onSince then
		t += os.clock() - stats.onSince
	end
	return t
end

local function formatTime(seconds)
	seconds = math.floor(seconds)
	return string.format("%02d:%02d:%02d",
		math.floor(seconds / 3600),
		math.floor((seconds % 3600) / 60),
		seconds % 60)
end

local function formatAgo(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s ago"
	elseif seconds < 3600 then
		return math.floor(seconds / 60) .. "m ago"
	end
	return math.floor(seconds / 3600) .. "h ago"
end

local function getTopFish()
	local bestName, bestCount = nil, 0
	for name, count in pairs(stats.perFish) do
		if count > bestCount then
			bestName, bestCount = name, count
		end
	end
	if bestName then
		return bestName .. " ×" .. bestCount
	end
	return "–"
end

	local t = getActiveTime()

	rows.harvests.Text = tostring(stats.harvests)

local function flash(lbl, color)
	lbl.TextColor3 = Color3.new(1, 1, 1)
	lbl.TextSize = 15
	TweenService:Create(lbl, TweenInfo.new(0.4, Enum.EasingStyle.Quad), {
		TextColor3 = color,
		TextSize = 12,
	}):Play()
end

local function registerCatch(fishName)
	fishName = fishName or "Fish"

	stats.catches += 1
	stats.streak += 1
	stats.perFish[fishName] = (stats.perFish[fishName] or 0) + 1

	table.insert(stats.recent, 1, { name = fishName, at = os.clock() })
	if #stats.recent > CONFIG.RecentCount then
		table.remove(stats.recent)
	end

	if stats.streak > stats.bestStreak then
		stats.bestStreak = stats.streak
		if stats.bestStreak > 1 then
			flash(rows.streak, COLORS.gold)
		end
	end

	flash(rows.catches, COLORS.good)
	updateBoard()
end

local function registerMiss()
	stats.missed += 1
	stats.streak = 0
	flash(rows.missed, COLORS.bad)
	updateBoard()
end

local function registerHarvest(name, category)
    category=category or FarmLogic.Category(name or "")
    if category and stats.resourceCounts[category]~=nil then
        stats.resourceCounts[category]+=1;stats.resourceLast[category]=name
    end
	stats.harvests += 1
	if name then
		stats.lastMineName = name
	end
	if stats.harvests % 10 == 0 then
		flash(rows.harvests, COLORS.harvest)
	end
	updateBoard()
end

    end
end

--------------------------------------------------------------------
-- New-item detection (used to name the fish you caught)
-- Rods moving between Backpack and Character are the same instance,
-- so only genuinely new instances count.
--------------------------------------------------------------------
local knownItems = setmetatable({}, { __mode = "k" })
local lastNewItem = nil -- { name = string, at = number }

local function markKnown(container)
	for _, child in ipairs(container:GetChildren()) do
		knownItems[child] = true
	end
end

local function onItemAdded(child)
	if not knownItems[child] then
		knownItems[child] = true
		lastNewItem = { name = child.Name, at = os.clock() }
	end
end

local function hookBackpack(backpack)
	markKnown(backpack)
	track(backpack.ChildAdded:Connect(onItemAdded))
end

local currentBackpack = player:FindFirstChildOfClass("Backpack")
if currentBackpack then

local lastFishingAim
local function clickOnce(action, myCycle)
    local camera = workspace.CurrentCamera
    if baitBusy or not camera or not hasRodEquipped() or not running(myCycle) then return false end
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local x, y = camera.ViewportSize.X * 0.7, camera.ViewportSize.Y * 0.6
    if action == "reel" and lastFishingAim then
        x, y = lastFishingAim.X, lastFishingAim.Y
    elseif root then
        local aim = root.Position + root.CFrame.LookVector * (CONFIG.CastDistance or 24) - Vector3.new(0, 4, 0)
        local point, visible = camera:WorldToScreenPoint(aim)
        if visible then x, y = point.X, point.Y end
    end
    local function overPanel(px, py)
        if not hub.Root.Visible then return false end
        local position, size = hub.Root.AbsolutePosition, hub.Root.AbsoluteSize
        return px >= position.X and px <= position.X + size.X and py >= position.Y and py <= position.Y + size.Y
    end
    if action ~= "reel" and overPanel(x, y) then x, y = camera.ViewportSize.X * 0.8, camera.ViewportSize.Y * 0.6 end
    if overPanel(x, y) then
        setStatus("Move the hub away from the water aiming area.", COLORS.warn)
        return false
    end
    -- Update the Mouse ray first; button events alone leave the aim over the UI.
    if action ~= "reel" or not lastFishingAim then
        local moved = pcall(function() VirtualInputManager:SendMouseMoveEvent(x, y, game) end)
        if not moved then return false end
        task.wait(0.03)
        if baitBusy or not running(myCycle) then return false end
        lastFishingAim = {X=x,Y=y}
    end
    fishingInputBusy = true
    local pressed = pcall(function() VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 0) end)
    local held = pcall(function()
        if not pressed then return end
        local deadline = os.clock() + (action == "cast" and CONFIG.CastHold or 0.01)
        while running(myCycle) and os.clock() < deadline do task.wait(0.01) end
    end)
    -- Release independently of both input errors and Stop during the charge.
    local released = pcall(function() VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 0) end)
    fishingInputBusy = false
    return pressed and held and released and running(myCycle)
end
local function getBobber()
    local folder = workspace:FindFirstChild("FishingBobbers")
    if not folder then return nil end
    local exact = folder:FindFirstChild(player.Name .. "Bobber")
    if exact and (exact:IsA("BasePart") or exact:IsA("Model")) then return exact end
    for _, child in ipairs(folder:GetChildren()) do
        local owner = child:GetAttribute("OwnerUserId") or child:GetAttribute("PlayerUserId")
        if owner == player.UserId and (child:IsA("BasePart") or child:IsA("Model")) then return child end
    end
    return nil
end

local function waitForTug(bobber, myCycle)
	local tugged = false
    local deadline = os.clock() + CONFIG.BiteTimeout

	local function checkTug()
		local value = bobber:GetAttribute("Tug")
		if value and value ~= false then
			tugged = true
		end
	end

	checkTug()
	local conn = bobber:GetAttributeChangedSignal("Tug"):Connect(checkTug)

	while running(myCycle) and bobber.Parent and not tugged and os.clock() < deadline do
		task.wait(CONFIG.FishingPollInterval)
	end

	conn:Disconnect()
	return tugged and running(myCycle), not tugged and bobber.Parent ~= nil and os.clock() >= deadline
end

-- Polls until predicate() is true, the cycle stops, or timeout. Returns predicate result.
local function waitUntil(myCycle, timeout, predicate)
	local deadline = os.clock() + timeout
	while running(myCycle) and os.clock() < deadline do
		if predicate() then
			return true
		end
		task.wait(CONFIG.FishingPollInterval)
	end
	-- A stopped/restarted cycle must not report a stale success after cancellation.
	return running(myCycle) and predicate()
end

-- Plant a crop at a CFrame (from Cobalt plant capture)
local function firePlant(cf, blockType)
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        local names = RemoteCatalog.SeedNames(blockType or CONFIG.PlantType)
        for _, tool in ipairs(RemoteCatalog.OwnedTools()) do
            if table.find(names, tool.Name) then
                if not RemoteCatalog.EquipTool(tool) then return false, "Could not equip seed tool" end
                break
            end
        end
    end
    return RemoteCatalog.CallPlacement({ cframe = cf, blockType = blockType or CONFIG.PlantType })
end

-- Plant a grid of crops around the player
local function plantAroundPlayer(blockType)
    if not RemoteCatalog.PlacementAvailable() then
        setStatus("Place one block manually with capture active first.", COLORS.warn)
        return 0
    end
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then
		return 0
	end
	blockType = blockType or CONFIG.PlantType
	local origin = hrp.Position
	local planted = 0
	local r = CONFIG.PlantRadius
	local step = CONFIG.PlantSpacing

	for x = -r, r, step do
		for z = -r, r, step do
			local pos = Vector3.new(origin.X + x, origin.Y, origin.Z + z)
			-- Keep upright rotation like the Cobalt sample
			local cf = CFrame.new(pos) * CFrame.Angles(0, 0, 0)
			if firePlant(cf, blockType) then
				planted += 1
			end
			task.wait(CONFIG.PlantDelay)
		end
	end
	return planted
end

-- Shared tool support from the Auto Wood Farmer.
-- Prefer an axe when available; otherwise fall back to a pickaxe.
local function equipBestResourceTool(category)
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end

    local preference = category == "wood" and woodToolPreference or mineToolPreference
    local wantAxe = category == "wood"
    local best, bestScore = nil, -math.huge
    for _, tool in ipairs(RemoteCatalog.OwnedTools()) do
        local name = string.lower(tool.Name)
        local isPick = string.find(name, "pickaxe", 1, true) ~= nil
        local isAxe = string.find(name, "axe", 1, true) ~= nil and not isPick
        if (wantAxe and isAxe) or (not wantAxe and isPick) then
            local score = 1
            if string.find(name, "mythic", 1, true) then score = 100
            elseif string.find(name, "diamond", 1, true) then score = 90
            elseif string.find(name, "gold", 1, true) then score = 80
            elseif string.find(name, "steel", 1, true) then score = 70
            elseif string.find(name, "iron", 1, true) then score = 60
            elseif string.find(name, "stone", 1, true) then score = 40
            elseif string.find(name, "wood", 1, true) then score = 20 end
            if not string.find(string.lower(preference), "auto best", 1, true) then
                local wanted = string.lower(preference):gsub(" axe", ""):gsub(" pickaxe", "")
                if not string.find(name, wanted, 1, true) then
                    score = -math.huge -- manual preference means exact material only
                else
                    score = score + 1000
                end
            end
            if score > bestScore then best, bestScore = tool, score end
        end
    end
    if best and bestScore > -math.huge then
        local ok = RemoteCatalog.EquipTool(best)
        local err = "inventory tool selection unavailable"
        if not ok then
            if setStatus then setStatus("Tool equip failed: " .. tostring(err), COLORS.bad) end
            return false
        end
        return true
    end
    if setStatus then
        setStatus((category == "wood" and "No matching axe: " or "No matching pickaxe: ") .. tostring(preference), COLORS.warn)
    end
    return false
end

local function equipPickaxe()

end

local function getPartPosition(obj)
	if obj:IsA("BasePart") then
		return obj.Position
	elseif obj:IsA("Model") then
        -- Model pivots may be far from their rendered rocks/crops.
        local boxed, bounds = pcall(function() return obj:GetBoundingBox() end)
        if boxed and bounds then return bounds.Position end
		local ok, pivot = pcall(function()
			return obj:GetPivot().Position
		end)
		if ok and pivot then
			return pivot
		end
	end
	local part = obj:FindFirstChildWhichIsA("BasePart", true)
	if part then
		return part.Position
	end
	return nil
end

local function targetCategory(name) return FarmLogic.Category(name) end
local function modeAllows(name)
    local category = targetCategory(name)
    if category == "crop" then return harvestEnabled and FarmLogic.CropMatches(name, CONFIG.HarvestCrop) end
    if category == "wood" then return woodEnabled end
    return category == "mine" and mineEnabled

local FarmState = { Cooldowns = setmetatable({}, { __mode = "k" }), FruitTrees = setmetatable({}, { __mode = "k" }) }
local function getMineTargets()
    local targets, seen = {}, {}
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return targets end
    local function consider(obj, allowCrops)
        if seen[obj] then return end
        seen[obj] = true
        local category = targetCategory(obj.Name)
        if not category or not modeAllows(obj.Name) then return end
        if category == "crop" and not allowCrops then return end
        if category == "wood" and allowCrops and harvestEnabled and CONFIG.FruitEnabled then
            if FarmState.FruitTrees[obj] or string.find(string.lower(obj.Name),"treeorange",1,true) or #FarmLogic.Fruits(obj)>0 then
                FarmState.FruitTrees[obj]=true;return
            end
        end
        if category == "crop" and not FarmLogic.IsReady(obj) then return end
        if (FarmState.Cooldowns[obj] or 0) > os.clock() then return end
        local position = getPartPosition(obj)
        if not position then return end
        local distance = (position - hrp.Position).Magnitude
        if distance > FarmLogic.Range(category, CONFIG) then return end
        table.insert(targets, {kind = category == "crop" and "model" or "block",
            instance = obj, name = obj.Name, distance = distance, category = category})
    end
    local function scan(folder, allowCrops)
        if folder then for _, obj in ipairs(folder:GetChildren()) do consider(obj, allowCrops) end end
    end
    local function scanResources(folder, depth)
        if not folder or depth < 0 then return end
        for _, obj in ipairs(folder:GetChildren()) do
            local category = targetCategory(obj.Name)
            local resource = category == "wood" or category == "mine"
            local containsResources = false
            if obj:IsA("Model") then
                for _, child in ipairs(obj:GetChildren()) do
                    local childCategory = targetCategory(child.Name)
                    if child:IsA("Model") and (childCategory == "wood" or childCategory == "mine") then containsResources = true; break end
                end
            end
            if resource and not containsResources and (obj:IsA("Model") or obj:IsA("BasePart")) then
                consider(obj, false)
            elseif not obj:IsA("BasePart") then
                scanResources(obj, depth - 1)
            end
        end
    end
    local prefabs = workspace:FindFirstChild("spawnPrefabs")
    if woodEnabled or mineEnabled then scanResources(prefabs, 5) end
    local owned = getPlayerIsland()
    local islands = workspace:FindFirstChild("Islands")
    if islands then
        for _, island in ipairs(islands:GetChildren()) do
            if island == owned or woodEnabled or mineEnabled then
                scan(island:FindFirstChild("Blocks"), island == owned)
                scan(island:FindFirstChild("Prefabs"), island == owned)
            end
        end
    end
    if harvestEnabled and CONFIG.FruitEnabled and owned then
        local blocks=owned:FindFirstChild("Blocks")
        if blocks then for _,tree in ipairs(blocks:GetChildren()) do
            for _,fruit in ipairs(FarmLogic.Fruits(tree)) do
                local position=getPartPosition(fruit)
                if position and not seen[fruit] and (FarmState.Cooldowns[fruit] or 0)<=os.clock() then
                    local distance=(position-hrp.Position).Magnitude
                    if distance<=CONFIG.CropRange then
                        seen[fruit]=true;FarmState.FruitTrees[tree]=true
                        table.insert(targets,{kind="fruit",instance=fruit,model=tree,name=tree.Name.." fruit",distance=distance,category="crop"})
                    end
                end
            end
        end end
    end
    table.sort(targets, function(a, b) return a.distance < b.distance end)
    return targets
end

-- RESOURCE AIM BEGIN
local function getInteractionPoint(target, origin)
    local best, distance
    local function consider(part)
        if not part:IsA("BasePart") or part.CanQuery == false or part.Transparency == 1 then return end
        local point=part.Position
        local ok,closest=pcall(function() return part:GetClosestPointOnSurface(origin) end)
        if ok and closest then point=closest end
        local magnitude=(point-origin).Magnitude
        if not distance or magnitude<distance then best,distance=point,magnitude end
    end
    if target:IsA("BasePart") then consider(target)
    else for _,part in ipairs(target:GetDescendants()) do consider(part) end end
    return best or getPartPosition(target)
end

-- The equipped tool supplies live request names and session data through its own client code.
local function resourceInputHit(target)
    local myCycle = harvestCycleId
    local function active()
        return gui.Parent ~= nil and harvestCycleId == myCycle and target.Parent ~= nil
            and modeAllows(target.Name) and (woodEnabled or mineEnabled or harvestEnabled) and not enabled
    end
    if enabled then return false, "Pause Fishing before using resource input" end
    if not active() then return false, "Resource action stopped" end
    if enabled then return false, "Pause Fishing before using resource input" end
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local tool = char and char:FindFirstChildOfClass("Tool")
    local category = targetCategory(target.Name)
    local toolName = tool and string.lower(tool.Name) or ""
    local pickaxe = string.find(toolName, "pickaxe", 1, true) ~= nil
    local axe = string.find(toolName, "axe", 1, true) ~= nil and not pickaxe
    if not root or category == "mine" and not pickaxe or category == "wood" and not axe then
        return false, "Wait for the matching axe/pickaxe to equip"
    end
    local position = getInteractionPoint(target, root.Position)
    if not position then return false, "Resource has no position" end
    local focused, textBox = pcall(function() return UserInputService:GetFocusedTextBox() end)
    if focused and textBox then return false, "Close chat or finish editing before resource input" end
    local reach = RemoteCatalog.GameModule("ReplicatedStorage.Shared.SharedConfig", "CLIENT_REACH")
    if type(reach) ~= "number" then reach = 20 end
    if (position - root.Position).Magnitude > reach then
        return false, "Move closer to " .. target.Name .. " (tool reach " .. reach .. " studs)"
    end
    local camera = workspace.CurrentCamera
    if not camera then return false, "Camera unavailable" end
    local point, visible = camera:WorldToScreenPoint(position)
    if not visible or point.Z <= 0 then return false, "Face " .. target.Name .. " so it is on screen" end
    local x, y = point.X, point.Y
    if hub.Root.Visible then
        local p, size = hub.Root.AbsolutePosition, hub.Root.AbsoluteSize
        if x >= p.X and x <= p.X + size.X and y >= p.Y and y <= p.Y + size.Y then
            return false, "Move the hub away from " .. target.Name
        end
    end
    local checked, objects = pcall(function() return playerGui:GetGuiObjectsAtPosition(x, y) end)
    if checked then
        for _, object in ipairs(objects) do
            if object:IsA("GuiButton") or object:IsA("TextBox") then
                return false, "Move the resource clear of on-screen controls"
            end
        end
    end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {char}
    local hit = workspace:Raycast(camera.CFrame.Position, position - camera.CFrame.Position, params)
    if not hit or hit.Instance ~= target and not hit.Instance:IsDescendantOf(target) then
        return false, "Resource is obstructed; move or turn toward " .. target.Name
    end
    local moved = pcall(function() VirtualInputManager:SendMouseMoveEvent(x, y, game) end)
    if not moved then return false, "Tool input unavailable in this environment" end
    task.wait(0.03)
    if not active() or category ~= "crop" and tool.Parent ~= char then return false, "Resource action stopped or tool changed" end
    local function input(down)
        if category == "crop" then VirtualInputManager:SendKeyEvent(down, Enum.KeyCode.F, false, game)
        else VirtualInputManager:SendMouseButtonEvent(x, y, 0, down, game, 0) end
    end
    local pressed = pcall(input, true)
    local held = pcall(function() if pressed then task.wait(category == "crop" and 0.1 or 0.06) end end)
    -- Always release even if Stop, UI cleanup or an input error happened during the press.
    local released = pcall(input, false)
    if not pressed or not held or not released then return false, "Tool input unavailable in this environment" end
    return active(), "Normal tool input"
end

local function harvestLoop(myCycle)
    local function active()
        return harvestCycleId == myCycle and (harvestEnabled or woodEnabled or mineEnabled) and gui.Parent ~= nil
    end
    while active() do
        local targets = getMineTargets()
        if #targets == 0 then
            setStatus("No matching ready crops on your island, or nearby wood/mining targets.", COLORS.muted)
            task.wait(CONFIG.MineScanInterval)
            continue
        end
        for _, target in ipairs(targets) do
            if not active() then break end
            local obj = target.instance
            if not obj.Parent then continue end
            if target.kind=="fruit" then
                if not harvestEnabled or not CONFIG.FruitEnabled or not target.model.Parent or not obj:IsDescendantOf(target.model) then continue end
            elseif not modeAllows(obj.Name) then continue end
            local char = player.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local position = getPartPosition(obj)
            if not hrp or not position or (position - hrp.Position).Magnitude > FarmLogic.Range(target.category, CONFIG) then continue end
            if target.category == "wood" or target.category == "mine" then
                if not equipBestResourceTool(target.category) then FarmState.Cooldowns[obj] = os.clock() + 2; continue end
            end
            if not active() then break end
            position = getPartPosition(obj)
            if not position or (position - hrp.Position).Magnitude > FarmLogic.Range(target.category, CONFIG) then continue end
            local before = FarmLogic.Snapshot(obj)
            if target.category == "crop" and not FarmLogic.IsReady(obj) then continue end
            local recorder = RemoteCatalog.Recorder
            local binding = recorder and recorder.HarvestBinding
            local learned = binding and binding.Remote.Parent and (binding.Target:IsA("Model") == obj:IsA("Model"))
            local nativeHarvest = target.category == "crop" and RemoteCatalog.HarvestEndpoint()
            local prompt = target.kind ~= "fruit" and target.category == "crop" and not learned and not nativeHarvest and FarmLogic.HarvestPrompt(obj) or nil
            local plantedFrame
            if obj:IsA("BasePart") then plantedFrame = obj.CFrame
            elseif obj:IsA("Model") then plantedFrame = obj:GetPivot() end
            local ok, failure
            if prompt then
                local promptPosition = getPartPosition(prompt.Parent) or position
                if (promptPosition - hrp.Position).Magnitude > prompt.MaxActivationDistance then continue end
                if stats.resourceAttempts then stats.resourceAttempts[target.category]+=1 end
                ok = FarmLogic.HoldPrompt(prompt, active, task.wait, os.clock)
                if not active() then break end
            elseif target.kind == "fruit" then
                if stats.resourceAttempts then stats.resourceAttempts.crop+=1 end
                ok,failure=RemoteCatalog.CallFruit(target.model,obj)
                if not ok then setStatus("Fruit picking waiting: "..tostring(failure),COLORS.warn) end
            elseif target.kind == "model" then
                if stats.resourceAttempts then stats.resourceAttempts[target.category]+=1 end
                ok, failure = fireModelHarvest(obj)
                if not ok then setStatus("Harvest waiting: " .. tostring(failure), COLORS.warn) end
            else
                if stats.resourceAttempts then stats.resourceAttempts[target.category]+=1 end
                ok, failure = fireBlockMine(obj)
                if not ok then setStatus("Mining waiting: " .. tostring(failure), COLORS.warn) end
            end
            local confirmed = false
            if ok then
                local deadline = os.clock() + 0.8
                repeat
                    if not active() then break end
                    confirmed = FarmLogic.Confirmed(obj, before, prompt)
                    if confirmed then break end
                    task.wait(0.05)
                until os.clock() >= deadline
            end
            if not active() then break end
            if ok and not confirmed then setStatus("No target change confirmed; retrying " .. target.name, COLORS.muted) end
            if confirmed then
                registerHarvest(target.name, target.category)
                setStatus("Confirmed: " .. target.name .. " ×" .. stats.harvests, COLORS.good)
                -- A plant that regrows in place must never be planted over.
                if target.category == "crop" and target.kind ~= "fruit" and not obj.Parent and plantedFrame and CONFIG.PlantEnabled and CONFIG.AutoReplant and not (RemoteCatalog.ServerAutoReplant or RemoteCatalog.ServerReplantEnabled and RemoteCatalog.ServerReplantEnabled()) then
                    if RemoteCatalog.PlacementAvailable() then
                        if active() then firePlant(plantedFrame, CONFIG.PlantType) end
                    else
                        setStatus("Harvest confirmed. Place one crop manually to enable replanting.", COLORS.warn)
                    end
                end
            end
            FarmState.Cooldowns[obj] = os.clock() + (ok and 0.25 or 2)
            task.wait(CONFIG.MineDelay)
        end
        -- All candidates may be out of prompt range; always yield before rescanning.
        task.wait(CONFIG.MineScanInterval)
    end
end

--------------------------------------------------------------------
-- Main fishing loop
--------------------------------------------------------------------
local setEnabled -- forward declaration
local setHarvestEnabled

local function run(myCycle)
	local failedCasts = 0
    local countedBites = setmetatable({}, {__mode = "k"})

	while running(myCycle) do
        if baitBusy then task.wait(0.05); continue end
		local bobber = getBobber()

		if bobber then
			failedCasts = 0
			setStatus("Waiting for fish...")
			local tugged, timedOut = waitForTug(bobber, myCycle)
			if not running(myCycle) then break end

			if timedOut then
                setEnabled(false)
                setStatus("No bite for " .. CONFIG.BiteTimeout .. "s. Reel manually, check water/bait, then restart.", COLORS.warn)
                return
            elseif not tugged then
				setStatus("Bobber disappeared")
				task.wait(0.1)
			elseif os.clock() - lastAction >= CONFIG.ActionCooldown then
				lastAction = os.clock()
				local reelTime = os.clock()
				if not countedBites[bobber] then stats.bites += 1; countedBites[bobber] = true end
				updateBoard()
				setStatus("Tug! Reeling in...", COLORS.gold)
                if not clickOnce("reel", myCycle) then
                    if not running(myCycle) then break end
                    setStatus("Reel input blocked; retrying.", COLORS.warn)
                    task.wait(0.25)
                    continue
                end
				task.wait(CONFIG.ReelSettleDelay)
                if not running(myCycle) then break end
				setStatus("Waiting for catch...")
				local gone = waitUntil(myCycle, CONFIG.ReelTimeout, function()
					return getBobber() == nil
				end)
				if not running(myCycle) then break end
				if gone then
					task.wait(0.05)
					local fishName
					if lastNewItem and lastNewItem.at >= reelTime then fishName = lastNewItem.name end
					registerCatch(fishName)
					setStatus("Caught " .. (fishName or "a fish") .. "! 🐟", COLORS.good)
				else
					registerMiss()
					setStatus("Reel not confirmed - retrying", COLORS.warn)
					task.wait(0.05)
				end
			end
		elseif not hasRodEquipped() then
			setStatus("Equip your fishing rod", COLORS.warn)
			task.wait(0.5)
		else
			if os.clock() - lastAction >= CONFIG.ActionCooldown then
				lastAction = os.clock()
				setStatus("Casting...")
				if clickOnce("cast", myCycle) then
					stats.casts += 1
					updateBoard()
					-- Only wait for a bobber after an actual cast input; this avoids
					-- burning the full timeout on frames where the cooldown is active.
					local found = waitUntil(myCycle, CONFIG.CastTimeout, function()
						return getBobber() ~= nil
					end)
					if not running(myCycle) then break end
					if not found then
						failedCasts += 1
						if failedCasts >= CONFIG.MaxFailedCasts then
							setEnabled(false)
							setStatus("Stopped: no bobber after " .. failedCasts .. " casts", COLORS.bad)
							return
						end
						setStatus(string.format("No bobber (%d/%d) - retrying", failedCasts, CONFIG.MaxFailedCasts), COLORS.warn)
						 task.wait(0.03)
					end
				else
                    if not running(myCycle) then break end
					setStatus("Input blocked - run in a supported client", COLORS.bad)
					task.wait(0.25)
				end
			else
				task.wait(0.02)
			end
		end
	end

	if cycleId == myCycle and not enabled then
		if not (harvestEnabled or woodEnabled or mineEnabled) then
			setStatus("Idle")
		end
	end
end

setEnabled = function(state)
	if state == enabled then
		return
	end

	enabled = state
	cycleId += 1

	if enabled then
		if not stats.onSince then
			stats.onSince = os.clock()
		end
		toggle.Text = "FISH ON  [" .. CONFIG.ToggleKey.Name .. "]"
		toggle.BackgroundColor3 = COLORS.on
		setStatus("Fishing started...")
        setBaitEnabled(true)
        task.spawn(function(token)
            local ok, failure = pcall(run, token)
            if not ok and token == cycleId then
                setEnabled(false)
                setStatus("Fishing stopped: " .. tostring(failure), COLORS.bad)
            end
        end, cycleId)
	else
        setBaitEnabled(false)
		toggle.Text = "FISH OFF  [" .. CONFIG.ToggleKey.Name .. "]"
		toggle.BackgroundColor3 = COLORS.off
		if not (harvestEnabled or woodEnabled or mineEnabled) and stats.onSince then
			stats.activeTime += os.clock() - stats.onSince
			stats.onSince = nil
		end
		if not (harvestEnabled or woodEnabled or mineEnabled) then
			setStatus("Idle")
		end
	end

	updateBoard()
end

local function refreshFarmButtons()
    harvestToggle.Text = harvestEnabled and "CROP FARM ON" or "CROP FARM OFF"
    harvestToggle.BackgroundColor3 = harvestEnabled and COLORS.on or COLORS.off
    woodToggle.Text = woodEnabled and "WOOD FARM ON" or "WOOD FARM OFF"
    woodToggle.BackgroundColor3 = woodEnabled and COLORS.on or COLORS.off
    mineToggle.Text = mineEnabled and "MINING ON" or "MINING OFF"
    mineToggle.BackgroundColor3 = mineEnabled and COLORS.on or COLORS.off
end

local function setFarmMode(mode, state)
    -- Resolve the requested mode explicitly. The old chained `and/or`
    -- expression could read another mode's state when this mode was off.
    local current
    if mode == "crop" then
        current = harvestEnabled
    elseif mode == "wood" then
        current = woodEnabled
    elseif mode == "mine" then
        current = mineEnabled
    else
        return
    end
    if current == state then return end
    if mode == "crop" then harvestEnabled = state
    elseif mode == "wood" then woodEnabled = state
    else mineEnabled = state end
    harvestCycleId += 1
    refreshFarmButtons()
    refreshModeStatus()
    updateBoard()
    if harvestEnabled or woodEnabled or mineEnabled then
        if not stats.onSince then stats.onSince = os.clock() end
        setStatus(mode == "crop" and "Crop farming started" or mode == "wood" and "Wood farming started" or "Mining started", COLORS.good)
        task.spawn(function(token)
            local ok, failure = pcall(harvestLoop, token)
            if not ok and token == harvestCycleId then
                harvestEnabled, woodEnabled, mineEnabled = false, false, false
                refreshFarmButtons()
                refreshModeStatus()
                setStatus("Farm stopped: " .. tostring(failure), COLORS.bad)
            end
        end, harvestCycleId)
    else
        if not enabled and stats.onSince then
            stats.activeTime += os.clock() - stats.onSince
            stats.onSince = nil
        end
        if not enabled then setStatus("Idle") end
    end
    refreshFarmButtons()
    refreshModeStatus()
    updateBoard()
    savePreferences()
end

setHarvestEnabled = function(state) setFarmMode("crop", state) end

-- Real expanding dropdowns matching the crop selector's visual style.
local function makeToolDropdown(page, anchor, options, kind, yPosition, onChoose)
    local menu = Instance.new("Frame")
    menu.Name = kind .. "Dropdown"
    menu.Size = UDim2.new(1, -12, 0, #options * 36)
    menu.Position = UDim2.fromOffset(6, yPosition)
    menu.BackgroundColor3 = COLORS.panel
    menu.BorderSizePixel = 0
    menu.Visible = false
    menu.ClipsDescendants = true
    menu.ZIndex = 25
    menu.Parent = page
    corner(menu, 6)
    local list = Instance.new("UIListLayout")
    list.Padding = UDim.new(0, 2)
    list.SortOrder = Enum.SortOrder.LayoutOrder
    list.Parent = menu
    for i, optionName in ipairs(options) do
        local opt = button({Parent = menu, Text = optionName, TextSize = 11,
            Size = UDim2.new(1, -4, 0, 34), Position = UDim2.new(), Bg = COLORS.panel})
        opt.LayoutOrder = i
        opt.ZIndex = 26
        track(opt.Activated:Connect(function()
            onChoose(i)
            anchor.Text = kind .. ": " .. optionName .. "  ▼"
            menu.Visible = false
        end))
    end
    track(anchor.Activated:Connect(function()
        menu.Visible = not menu.Visible
    end))
end
